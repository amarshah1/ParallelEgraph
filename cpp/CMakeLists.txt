cmake_minimum_required(VERSION 3.20)
project(parallel_egraph_cpp LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_compile_options(-Wall -Wextra -Wpedantic -Wno-unused-parameter)

# ---- parlay ----
include(FetchContent)
FetchContent_Declare(
  parlaylib
  GIT_REPOSITORY https://github.com/cmuparlay/parlaylib.git
  GIT_TAG        master
)
# Prevent parlay from adding its own install / test targets.
set(PARLAY_TEST OFF CACHE BOOL "" FORCE)
set(PARLAY_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PARLAY_BENCHMARK OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(parlaylib)

# ---- CaDiCaL (SAT solver, IPASIR-UP API) -----------------------------------
# CaDiCaL ships with a hand-rolled ./configure + makefile.in build, not CMake.
# We FetchContent the sources, then build ourselves as a static lib by
# globbing src/*.cpp. Keeps everything in one CMake graph; no nested make
# invocation. We define VERSION / IDENTIFIER / COMPILER / FLAGS / DATE
# manually because CaDiCaL's source generator script normally fills them in.
FetchContent_Declare(
  cadical
  GIT_REPOSITORY https://github.com/arminbiere/cadical.git
  GIT_TAG        master
  # CaDiCaL has no CMakeLists.txt; MakeAvailable still fetches but skips
  # the (nonexistent) add_subdirectory step. We build it ourselves below.
)
FetchContent_MakeAvailable(cadical)

file(GLOB CADICAL_SRCS CONFIGURE_DEPENDS
  "${cadical_SOURCE_DIR}/src/*.cpp"
  "${cadical_SOURCE_DIR}/src/*.c"   # kitten.c (sub-solver used by sweep)
)
# Exclusions:
#   mockup.cpp  — stand-alone harness, not library API
#   cadical.cpp — has main(), the CLI binary, not us
list(FILTER CADICAL_SRCS EXCLUDE REGEX "/mockup\\.cpp$")
list(FILTER CADICAL_SRCS EXCLUDE REGEX "/cadical\\.cpp$")
# We do *not* exclude version.cpp — it defines CaDiCaL::identifier()/etc
# referenced from solver.cpp's build() banner. We only need to make sure
# build.hpp exists with the right macros for it to compile.
# (Original exclusion left as a comment for clarity:)
# list(FILTER CADICAL_SRCS EXCLUDE REGEX "/version\\.cpp$")

# Synthesize a build.hpp the way CaDiCaL's configure normally would.
file(READ "${cadical_SOURCE_DIR}/VERSION" CADICAL_VERSION)
string(STRIP "${CADICAL_VERSION}" CADICAL_VERSION)
file(WRITE "${cadical_SOURCE_DIR}/src/build.hpp"
"// Auto-generated by parallel-egraph CMake (replaces ./configure output).
#define VERSION \"${CADICAL_VERSION}\"
#define IDENTIFIER \"parallel-egraph-build\"
#define COMPILER \"${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}\"
#define BUILD \"cmake-fetchcontent\"
#define DATE __DATE__
#define FLAGS \"\"
")

# CaDiCaL's resources.cpp/file.cpp call closefrom(3), a Linux/BSD-only libc
# extension. Patch file.cpp to skip the call on platforms where it's missing.
# We do this with a one-shot file edit (idempotent: re-running CMake just
# rewrites the same content). Done inline rather than as a CaDiCaL source
# patch because there's exactly one site and it's a no-op fall-through.
file(READ "${cadical_SOURCE_DIR}/src/file.cpp" CADICAL_FILE_CPP)
string(REPLACE "::closefrom (3);"
               "/* parallel-egraph patch: closefrom() not available on macOS */"
               CADICAL_FILE_CPP_PATCHED "${CADICAL_FILE_CPP}")
if(NOT "${CADICAL_FILE_CPP}" STREQUAL "${CADICAL_FILE_CPP_PATCHED}")
  file(WRITE "${cadical_SOURCE_DIR}/src/file.cpp" "${CADICAL_FILE_CPP_PATCHED}")
endif()

add_library(cadical STATIC ${CADICAL_SRCS})
target_include_directories(cadical PUBLIC ${cadical_SOURCE_DIR}/src)
target_compile_definitions(cadical PRIVATE
  NUNLOCKED   # plain stdio (avoids unlocked_putc on macOS)
  NTRACING    # disable proof tracing
)
# CaDiCaL is not warning-clean; suppress only on its target.
target_compile_options(cadical PRIVATE -w)

# ---- core library ----
add_library(parallel_egraph_core STATIC
  src/unionfind.cpp
  src/smtlib.cpp
  src/egraph.cpp
  src/sat_solve.cpp
  src/solve.cpp
)
target_include_directories(parallel_egraph_core
  PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include
)
target_link_libraries(parallel_egraph_core PUBLIC parlay cadical)

# ---- main CLI ----
add_executable(parallel-egraph src/main.cpp)
target_link_libraries(parallel-egraph PRIVATE parallel_egraph_core)

# ---- benchmark ----
add_executable(rebuild_compare_bench bench/rebuild_compare.cpp)
target_link_libraries(rebuild_compare_bench PRIVATE parallel_egraph_core)

# ---- tests ----
enable_testing()
add_executable(unionfind_test tests/unionfind_test.cpp)
target_link_libraries(unionfind_test PRIVATE parallel_egraph_core)
add_test(NAME unionfind COMMAND unionfind_test)

add_executable(regression_test tests/regression_test.cpp)
target_link_libraries(regression_test PRIVATE parallel_egraph_core)
target_compile_definitions(regression_test PRIVATE
  TESTS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../tests"
)
add_test(NAME regression COMMAND regression_test)

add_executable(parallel_close_test tests/parallel_close_test.cpp)
target_link_libraries(parallel_close_test PRIVATE parallel_egraph_core)
target_compile_definitions(parallel_close_test PRIVATE
  SYNTHETIC_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../synthetic_benchmarks"
)
add_test(NAME parallel_close COMMAND parallel_close_test)

add_executable(bool_regression_test tests/bool_regression_test.cpp)
target_link_libraries(bool_regression_test PRIVATE parallel_egraph_core)
target_compile_definitions(bool_regression_test PRIVATE
  BOOL_TESTS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/bool"
)
add_test(NAME bool_regression COMMAND bool_regression_test)

add_executable(concurrent_append_vec_test tests/concurrent_append_vec_test.cpp)

add_test(NAME concurrent_append_vec COMMAND concurrent_append_vec_test)
